## [C](https://atcoder.jp/contests/abc308/tasks/abc308_c)
$A_i/(A_i+B_i) < A_j/(A_j+B_j) 순서대로 sort하는 문제인데 한번 식정리 해줘야함
```math
A_i \times (A_j+B_j) < A_j \times (A_i + B_i) -> A_iB_j < A_jB_i
```
로 식정리 가능하다.

## [E](https://atcoder.jp/contests/abc308/tasks/abc308_e)
  M을 iterate 하고 suffix 2개로 하니까 겹쳐서 못풀었는데 
  E를 iterate 하면 prefix,suffix 로 나눌수있어서 독립되게 볼수있다.
  이는 9*O(n)으로 가능하다.
  ```cpp
  void solve() {
      cin>>n; a.resize(n);
      for(auto &i : a) cin>>i;
      cin>>s;
      vector pref(n,vi(3));
      vector suf(n,vi(3));
      for(int i=0;i<n;i++){
          if(i>0) pref[i] = pref[i-1];
          if(s[i] == 'M'){
              pref[i][a[i]]++;
          }
      }
      for(int i=n-1;i>=0;i--){
          if(i<n-1) suf[i] = suf[i+1];
          if(s[i] == 'X'){
              suf[i][a[i]]++;
          }
      }
      ll ret = 0;
      for(int i=0;i<n;i++){
          if(s[i] != 'E') continue;
          for(int j=0;j<3;j++){
              for(int k=0;k<3;k++){
                  ret += pref[i][j] * suf[i][k] * get_mex(a[i],j,k);
              }
          }
      }
      cout<<ret<<endl;
  }
  ```
## [F](https://atcoder.jp/contests/abc308/tasks/abc308_f)

```cpp
void solve() {
    cin>>n>>m; vi a(n),l(m),d(m);
    priority_queue<ll> Q;
    for(auto &i : a) cin>>i;
    for(auto &i : l) cin>>i;
    for(auto &i : d) cin>>i;
    sort(all(a));
    vector<pi> b(m);
    for(int i=0;i<m;i++) b[i] = {l[i],d[i]};
    sort(all(b));
    int i = 0;
    ll ret = 0;
    for(auto x : a){
        while(i < m and b[i].first <= x) Q.emplace(b[i++].second);
        if(Q.size()){
            x -= Q.top(); Q.pop();
        }
        ret += x;
    }
    cout<<ret<<endl;
}
```
