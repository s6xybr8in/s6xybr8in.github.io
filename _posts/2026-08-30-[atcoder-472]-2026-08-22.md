## [D](https://atcoder.jp/contests/abc472/tasks/abc472_d) 00 : 59 : 28
문제 지문을 잘못읽어서 시간이 많이 걸렸다.
문제를 제대로 읽어보니 mutli source bfs로 푸는 문제였다.
```cpp
vector<string> a;
vector<vector<bool>> vis; 
int n,m,k;
int dx[] = {0,0,1,-1};
int dy[] = {1,-1,0,0};
void solve() {
    cin>>n>>m>>k;
    a.resize(n);
    vi row(n),col(m);
    vis = vector(n,vector<bool>(m));
    for(auto &i : a) cin>>i;


    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            if(a[i][j] == '#'){
                row[i] = 1;
                col[j] = 1; 
            }
        }
    }
    queue<pi> Q;
    int ret = 0;
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++) if(a[i][j] == '.' && row[i] == 0 && col[j] == 0) {
            Q.emplace(0,i*m+j);
            // cout<<i+1<<" "<<j+1<<endl;
            ret++;
            vis[i][j] = true; 
        }
    }
    while(Q.size()){
        auto [step,d] = Q.front(); Q.pop();
        int x = d/m; 
        int y = d%m;
        // cout<<x+1<<' '<<y+1<<endl;
        if(step == k) continue;
        for(int i=0;i<4;i++){
            int nx = x + dx[i];
            int ny = y + dy[i]; 
            if(nx < 0 || ny < 0 || nx > n-1 || ny > m-1 || a[nx][ny] == '#' || vis[nx][ny]) continue;
            Q.emplace(step+1,nx*m+ny);
            vis[nx][ny] = true;
            ret++;
        }
    
    }
    cout<<ret<<endl;
}
```
## [E](https://atcoder.jp/contests/abc472/tasks/abc472_e) Upsolving
음 일단 사이클인지 dfs로 판단하는 방법은 방문순서에 따라 번호를 붙히고, 순서차이가 2이상이라면 이는 사이클이다.
그러면 순서차이가 짝수라면 홀수 사이클임은 쉽게 알수있다. 
사이클이 그러면 여러개가 겹쳐 있을 수 있는데, 짝수사이클은 차이가 항상 홀수라서 상관없다. 
홀수사이클 + 짝수 사이클, 홀수 사이클 + 홀수 사이클이 문제인데 이는 먼저 사이클 여부를 확인하고 방문을 하면 해결 할 수 있다.

## 정답 코드

```cpp
vi G[MAX]; 
bool vis[MAX];
int num[MAX];
int n,m;
int flag;
int cnt;
int start,ed;
void dfs(int x){
    vis[x] = true;
    num[x] = ++cnt;
    for(auto nx : G[x]){ // 사이클 여부 확인
        if(flag) return;
        if(vis[nx]){
            int delta = num[x] - num[nx];
            if(delta == 1) continue;
            if(delta%2 == 0){
                flag = true;
                start = num[nx];
                ed = num[x];
                return;
            }
        }
    }
    if(flag) return;

    for(auto nx : G[x]){ // 사이클 여부 확인 후 탐색
        if(flag) return;
        if(vis[nx]) continue;
        dfs(nx);
    }
    if(flag) return;
    num[x] = 0;
    cnt--;
}


void solve() {
    cin>>n>>m;
    while(m--){
        int u,v; cin>>u>>v;
        G[u].emplace_back(v);
        G[v].emplace_back(u);
    }
    for(int i=1;i<=n;i++){
        if(flag) break;
        if(vis[i]) continue;
        cnt = 0;
        dfs(i);
        // cout<<endl;
    }
    
    if(flag){
        int k = ed - start + 1;
        // cout<<"K"<<k<<endl;
        vector<int> ret(k+1);
        for(int i=1;i<=n;i++){
            int x = num[i] - start + 1;
            if(1 <= x && x <= k) ret[x] = i;
        }
        cout<<k<<endl;
        for(int i=1;i<=k;i++) cout<<ret[i]<<" ";
        cout<<endl;
    }else{
        cout<<-1<<endl;
    }
    
    
    
    for(int i=1;i<=n;i++){
        G[i].clear();
        num[i] = vis[i] = 0;
    }
    flag = 0;
    cnt = 0;
    
}
```
